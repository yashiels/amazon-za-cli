def java_bytes(s):
    out = []
    for tok in s.split(","):
        tok = tok.strip()
        if tok.startswith("Ascii."):
            out.append(ASCII[tok[6:]])
        else:
            out.append(int(tok) & 0xFF)
    return bytes(out)

def sign(keyname, method, path, query, headers):
    kid, secret, alg = KEYS[keyname]
    t = time.strftime("%Y-%m-%dT%H:%M:%S+0000", time.gmtime())
    s = f"{method}\n{t}\n{kid}\n{alg}\n{path}\n{query}\n"
    names = sorted(k.lower() for k in headers)
    low = {k.lower(): v for k, v in headers.items()}
    for n in names:
        s += f"{n}:{low[n]}\n"
    sh = ";".join(names)
    s += sh
    dig = hashlib.sha1 if alg == "HmacSHA1" else hashlib.sha256
    return f"{alg} Time={t} SignedHeaders={sh} Signature={hmac.new(secret, s.encode(), dig).hexdigest()}/{kid}"


def get(keyname, path, typ, expands=(), query=""):
    accept = f'application/vnd.com.amazon.api+json; type="{typ}"'
    if expands:
        accept += f'; expand="{",".join(expands)}"'
    h = {"Accept-Language": "en-ZA", "Accept": accept, "X-Amzn-Session-Id": sid, "X-Amzn-UBID": ubid}
    h["X-AAPI-Signature"] = sign(keyname, "GET", path, query, h)
    h["User-Agent"] = UA
    url = "https://data.amazon.co.za" + path + (("?" + query) if query else "")
    try:
        r = opener.open(urllib.request.Request(url, headers=h))
        return r.status, r.read().decode()
    except urllib.error.HTTPError as e:
        return e.code, e.read().decode()
